cmake_minimum_required(VERSION 3.16)
project(amr_mission LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
foreach(dep rclcpp rclcpp_action amr_interfaces nav2_msgs nav_msgs sensor_msgs geometry_msgs std_msgs std_srvs diagnostic_msgs tf2 tf2_ros ament_index_cpp yaml_cpp_vendor)
  find_package(${dep} REQUIRED)
endforeach()
find_package(yaml-cpp REQUIRED)
add_library(mission_core src/mission_core.cpp)
target_include_directories(mission_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
add_library(station_registry src/station_registry.cpp)
target_link_libraries(station_registry PUBLIC mission_core yaml-cpp)
add_executable(validate_stations src/validate_stations.cpp)
target_link_libraries(validate_stations station_registry)
add_executable(mission_manager src/mission_node.cpp)
target_link_libraries(mission_manager mission_core station_registry)
ament_target_dependencies(mission_manager rclcpp rclcpp_action amr_interfaces nav2_msgs nav_msgs sensor_msgs geometry_msgs std_msgs std_srvs diagnostic_msgs tf2 tf2_ros ament_index_cpp)
add_executable(motion_guard src/motion_guard.cpp)
ament_target_dependencies(motion_guard rclcpp geometry_msgs sensor_msgs nav_msgs std_msgs std_srvs)
install(TARGETS mission_core station_registry validate_stations mission_manager motion_guard EXPORT export_amr_mission ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
file(GLOB scripts LIST_DIRECTORIES false CONFIGURE_DEPENDS scripts/*)
if(scripts)
  install(PROGRAMS ${scripts} DESTINATION lib/${PROJECT_NAME})
endif()
if(BUILD_TESTING)
  enable_testing()
  add_executable(core_tests test/core_tests.cpp)
  target_link_libraries(core_tests mission_core)
  add_test(NAME core_tests COMMAND core_tests)
  add_executable(registry_tests test/registry_tests.cpp)
  target_link_libraries(registry_tests station_registry)
  add_test(NAME registry_tests COMMAND registry_tests ${CMAKE_CURRENT_SOURCE_DIR}/config/stations.yaml ${CMAKE_CURRENT_SOURCE_DIR}/../amr_navigation/maps/warehouse_v1.yaml)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(ros_integration test/test_ros_integration.py TIMEOUT 180)
  ament_add_pytest_test(motion_guard test/test_motion_guard.py TIMEOUT 40)
  ament_add_pytest_test(log_failure test/test_log_failure.py TIMEOUT 20)
endif()
ament_export_targets(export_amr_mission HAS_LIBRARY_TARGET)
ament_export_include_directories(include)
ament_package()
