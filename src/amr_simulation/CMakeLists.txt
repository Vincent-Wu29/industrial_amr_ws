cmake_minimum_required(VERSION 3.16)
project(amr_simulation)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(tf2_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(gz_transport_vendor REQUIRED)
find_package(gz_msgs_vendor REQUIRED)
find_package(gz-transport13 REQUIRED)
find_package(gz-msgs10 REQUIRED)
find_package(gz_sim_vendor REQUIRED)
find_package(gz_plugin_vendor REQUIRED)
find_package(gz-sim8 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
add_executable(gz_truth_bridge src/gz_truth_bridge.cpp)
target_compile_features(gz_truth_bridge PRIVATE cxx_std_17)
target_compile_options(gz_truth_bridge PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(gz_truth_bridge rclcpp tf2_msgs geometry_msgs)
target_link_libraries(gz_truth_bridge gz-transport13::gz-transport13 gz-msgs10::gz-msgs10)
install(TARGETS gz_truth_bridge DESTINATION lib/${PROJECT_NAME})
add_library(amr_truth_system SHARED src/truth_system.cpp)
target_compile_features(amr_truth_system PRIVATE cxx_std_17)
target_compile_options(amr_truth_system PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(amr_truth_system gz-sim8::gz-sim8 gz-plugin2::gz-plugin2-register gz-transport13::gz-transport13 gz-msgs10::gz-msgs10)
install(TARGETS amr_truth_system LIBRARY DESTINATION lib)
foreach(resource launch config maps worlds urdf behavior_trees)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${resource}")
    install(DIRECTORY ${resource} DESTINATION share/${PROJECT_NAME})
  endif()
endforeach()
file(GLOB scripts LIST_DIRECTORIES false CONFIGURE_DEPENDS scripts/*)
if(scripts)
  install(PROGRAMS ${scripts} DESTINATION lib/${PROJECT_NAME})
endif()
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  file(GLOB tests test/test_*.py)
  foreach(test ${tests})
    get_filename_component(name ${test} NAME_WE)
    ament_add_pytest_test(${name} ${test} TIMEOUT 120)
  endforeach()
endif()
ament_package()
