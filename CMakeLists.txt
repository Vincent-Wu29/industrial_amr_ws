cmake_minimum_required(VERSION 3.16)
project(industrial_amr_domain LANGUAGES CXX)
enable_testing()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(amr_core src/amr_mission/src/mission_core.cpp)
target_include_directories(amr_core PUBLIC src/amr_mission/include)
if(MSVC)
  target_compile_options(amr_core PRIVATE /W4)
else()
  target_compile_options(amr_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(core_tests src/amr_mission/test/core_tests.cpp)
target_link_libraries(core_tests PRIVATE amr_core)
add_test(NAME mission_core COMMAND core_tests)
